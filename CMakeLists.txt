cmake_minimum_required(VERSION 3.22)
option(BUILD_SIMULATOR "Build the Boost.Beast Udacity simulator bridge" ON)
if(BUILD_SIMULATOR)
  list(APPEND VCPKG_MANIFEST_FEATURES simulator)
endif()
project(MPC_Learning LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(mpc_core src/MPC.cpp src/Controller.cpp)
target_include_directories(mpc_core PUBLIC src)
include(CTest)
if(BUILD_SIMULATOR)
  find_package(Boost CONFIG REQUIRED COMPONENTS beast)
  find_package(nlohmann_json CONFIG REQUIRED)
  find_package(Threads REQUIRED)
  add_library(mpc_adapter src/simulator_session.cpp src/simulator_connection.cpp)
  target_link_libraries(mpc_adapter PUBLIC mpc_core Boost::beast nlohmann_json::nlohmann_json Threads::Threads)
  if(WIN32)
    target_compile_definitions(mpc_adapter PUBLIC _WIN32_WINNT=0x0601)
  endif()
  add_executable(mpc src/main.cpp)
  target_link_libraries(mpc PRIVATE mpc_adapter)
  if(BUILD_TESTING)
    add_executable(simulator_test tests/simulator_test.cpp)
    target_link_libraries(simulator_test PRIVATE mpc_adapter)
    add_test(NAME simulator_protocol COMMAND simulator_test)
    set_tests_properties(simulator_protocol PROPERTIES TIMEOUT 20)
  endif()
endif()
if(BUILD_TESTING)
  add_executable(mpc_test tests/mpc_test.cpp)
  target_link_libraries(mpc_test PRIVATE mpc_core)
  add_test(NAME mpc_checks COMMAND mpc_test)
endif()

add_executable(compare_controllers src/compare_controllers.cpp)
target_link_libraries(compare_controllers PRIVATE mpc_core)
if(BUILD_TESTING)
  add_executable(controller_test tests/controller_test.cpp)
  target_link_libraries(controller_test PRIVATE mpc_core)
  add_test(NAME controller_checks COMMAND controller_test)
endif()
